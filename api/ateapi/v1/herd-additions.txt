# Fields herd adds to the upstream messages of the subset (internal/cmd/protosubset -additions; ADR 0001, PLT-330).
# Each block's lines are appended to the subset's message of the same name. herd's field numbers start at 1000, far
# from upstream's, so a field upstream adds later never takes one. A real Substrate ignores these fields (proto3
# unknown fields).

message CreateActorTemplateRequest {
  // herd addition (PLT-330): a one-shot registry credential for the
  // template's images, used only for the pull that builds the template's
  // golden snapshot and taking precedence over the node's own registry login
  // (herdlet -registry-auth) for the hosts it names. A docker config file's
  // `auths` as JSON, the herdlet -registry-auth format:
  //   {"auths": {"ghcr.io": {"auth": "<base64 of user:password>"}}}
  // or {"username", "password"}, or {"identitytoken"} per host. It is held in
  // herd-api's memory until the golden build ends, passed to the herdlet that
  // builds it, never stored, never logged, and never returned: no
  // ActorTemplate field carries it, so GetActorTemplate cannot.
  //
  // +k8s:optional
  string image_pull_auth = 1000;

  // herd addition (DEFAUL-1600): the hosts of the template's golden build pull
  // that speak plain HTTP rather than https, each a host[:port] matched exactly
  // as written: an image store run without TLS and its token realm. One-shot
  // like image_pull_auth: used only for the pull that builds the template's
  // golden snapshot, beside the node's own -insecure-registry hosts, never
  // stored and never returned. A login (image_pull_auth or the node's) is sent
  // over plain HTTP only to a host named here or by the node.
  //
  // +k8s:optional
  repeated string image_plain_http_hosts = 1001;
}
