# Fields herd adds to the upstream messages of the subset (internal/cmd/protosubset -additions; ADR 0001, PLT-330).
# Each block's lines are appended to the subset's message of the same name. herd's field numbers start at 1000, far
# from upstream's, so a field upstream adds later never takes one. A real Substrate ignores these fields (proto3
# unknown fields).

message CreateActorTemplateRequest {
  // herd addition (PLT-330): a one-shot registry credential for the
  // template's images, used only for the pull that builds the template's
  // golden snapshot and taking precedence over the node's own registry login
  // (herdlet -registry-auth) for the hosts it names. A docker config file's
  // `auths` as JSON, the herdlet -registry-auth format:
  //   {"auths": {"ghcr.io": {"auth": "<base64 of user:password>"}}}
  // or {"username", "password"}, or {"identitytoken"} per host. It is held in
  // herd-api's memory until the golden build ends, passed to the herdlet that
  // builds it, never stored, never logged, and never returned: no
  // ActorTemplate field carries it, so GetActorTemplate cannot.
  //
  // +k8s:optional
  string image_pull_auth = 1000;

  // herd addition (DEFAUL-1600): the hosts of the template's golden build pull
  // that speak plain HTTP rather than https, each a host[:port] matched exactly
  // as written: an image store run without TLS and its token realm. One-shot
  // like image_pull_auth: used only for the pull that builds the template's
  // golden snapshot, beside the node's own -insecure-registry hosts, never
  // stored and never returned. A login (image_pull_auth or the node's) is sent
  // over plain HTTP only to a host named here or by the node.
  //
  // +k8s:optional
  repeated string image_plain_http_hosts = 1001;
}

message CreateActorRequest {
  // herd addition (PLT-405): a one-shot registry credential for the actor's
  // images, in image_pull_auth's format on CreateActorTemplateRequest and
  // checked the same way, used only if this create boots the actor cold (no
  // golden snapshot, or one no worker can restore, such as a MEMORY snapshot
  // of another restore family) and taking precedence over the node's own
  // registry login for the hosts it names. It is held in herd-api's memory for
  // that boot only, passed to the herdlet that boots it, never stored, never
  // logged, never returned and never used by a later boot of the actor.
  //
  // +k8s:optional
  string image_pull_auth = 1000;

  // herd addition (PLT-405): the plain-HTTP registry hosts of that cold
  // boot's pull, as image_plain_http_hosts on CreateActorTemplateRequest, and
  // one-shot like image_pull_auth above.
  //
  // +k8s:optional
  repeated string image_plain_http_hosts = 1001;
}

message Actor {
  // herd addition (PLT-489): opaque bytes the creator attaches at
  // CreateActor, at most 4096, stored verbatim and returned by GetActor and
  // ListActors. Set once: an UpdateActor that carries other bytes here is
  // refused InvalidArgument, and one that carries none leaves them as they
  // are. herd never reads them; Everdict stores its provision grant here (a
  // JWS the control plane signed after the session row was durable).
  //
  // +k8s:optional
  // +k8s:maxLength=4096
  // +k8s:immutable
  bytes provenance = 1000;
}

message ActorStatus {
  // herd addition (PLT-489): who created the Actor, as herd-api
  // authenticated the CreateActor call: the caller's SPIFFE ID
  // ("spiffe://<trust domain>/<path>"), or "hs256:<sub>" for a break-glass
  // HS256 bearer. Set by the server at CreateActor and immutable; whatever a
  // request carries here is ignored. Empty for an actor created before
  // PLT-489. A write on the actor (delete, suspend, resume, tag, egress
  // policy, exec through the router) is allowed to this caller or to an
  // operator (ADR 0001 A-60).
  //
  // +k8s:optional
  string creator = 1000;
}
